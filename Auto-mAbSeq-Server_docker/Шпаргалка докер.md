# Комманды для запуска докера

# 1. Остановить и удалить контейнеры
docker-compose down

# 2. Удалить старый образ (опционально)
docker rmi auto-mabseq-server_docker_auto-mabseq:latest

# 3. Очистить систему Docker
docker system prune -f

# 4. Пересобрать
docker-compose build --no-cache

# 5. Запустить
docker-compose up -d


# Когда контрейнер запущен
# 1. Статус контейнера
docker-compose ps

# 2. Логи Flask
docker-compose logs -f

# 3. Health check
curl http://localhost:5000/api/health

# 4. Главная страница
curl http://localhost:5000/
Если всё работает:

# Ссылка в браузере
http://localhost:5000/  

# После завершения пайплайна, если что-то пошло не так и остались данные. Они будут изолированы, так как для каждой задачи свой job, но могут забивать кэш
# Зайти в контейнер как bash
docker exec -it Auto-mAbSeq-app bash
docker exec Auto-mAbSeq-app ls -lah /data/

# Внутри контейнера:
rm -rf /data/raw/sanger/*
rm -rf /data/intermediate/*
rm -rf /data/results/*
rm -rf /data/logs/*
rm -rf /data/jobs/*

# Выйти
exit
